Usage Guide

Eloquent Mode (Standard)
@livewire('aftable', [
    'model'   => App\Models\User::class,
    'columns' => [
        ['key' => 'id',    'label' => 'ID'],
        ['key' => 'name',  'label' => 'Name'],
        ['key' => 'email', 'label' => 'Email'],
        // Blade template column:
        ['key' => 'status', 'label' => 'Status',
         'raw' => '<span class="badge bg-success">{{ $row->status }}</span>'],
        // Closure column (fastest — no Blade overhead):
        ['key' => 'price', 'label' => 'Price',
         'raw' => fn($row) => '<b>'.number_format($row->price, 2).'</b>'],
    ],
    'sortBy'  => 'created_at',
    'sort'    => 'desc',
])
Static / Foreach Mode NEW

Pass any PHP array or Collection as 'data'. No Eloquent model required. Great for API responses, computed data, or small datasets.

@php
    $employees = [
        ['name' => 'Alice', 'salary' => 5000, 'dept' => 'Engineering'],
        ['name' => 'Bob',   'salary' => 4500, 'dept' => 'Design'],
    ];
@endphp

@livewire('aftable', [
    'data'    => $employees,    // array or Collection — no model needed
    'vars'    => ['currency' => 'USD'],   // extra vars for raw templates
    'columns' => [
        ['key' => 'name',   'label' => 'Name'],
        ['key' => 'dept',   'label' => 'Department'],
        ['key' => 'salary', 'label' => 'Salary', 'raw' =>
            '{{ $currency }} {{ number_format($row->salary, 2) }}'],
    ],
])
Custom Table Template NEW

Use your app's existing table theme by passing a custom <table> tag. The component injects its headers and rows automatically.

{{-- Blade directive syntax --}}
@aftable(['model' => App\Models\User::class, 'columns' => [...]])
    <table class="table table-hover table-striped my-theme-table">
        <thead class="bg-dark text-white"><tr></tr></thead>
        <tbody></tbody>
    </table>
@endaftable

{{-- Or pass customTemplate directly --}}
@livewire('aftable', [
    'model'          => App\Models\User::class,
    'columns'        => [...],
    'customTemplate' => '<table class="table table-hover my-theme"></table>',
])
Closure Raw Columns

Use a PHP Closure as 'raw' for complex logic without Blade overhead. Receives $row and $vars.

'columns' => [
    // Simple closure — $row is the current row
    ['key' => 'status', 'label' => 'Status',
     'raw' => fn($row) => $row->active
         ? '<span class="badge bg-success">Active</span>'
         : '<span class="badge bg-danger">Inactive</span>'],

    // Closure with extra vars ($vars is your 'vars' array)
    ['key' => 'price', 'label' => 'Price',
     'raw' => fn($row, $vars) => e($vars['currency']).' '.number_format($row->price, 2)],

    // Dynamic CSS classes (no eval() — uses closure or key:value shorthand)
    ['key' => 'status', 'label' => 'Status',
     'classCondition' => [
         'text-success fw-bold' => fn($row) => $row->active,
         'text-danger'          => 'status:inactive',   // key:value shorthand
         'opacity-50'           => '!active',           // negation shorthand
     ]],
]
